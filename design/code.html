  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  siswa_id UUID REFERENCES siswa(id),
  mata_pelajaran_id UUID REFERENCES mapel(id),
  semester VARCHAR(10) DEFAULT 'GENAP_2024_2025',
  tp_1_score NUMERIC(5,2),
  tp_2_score NUMERIC(5,2),
  tp_3_score NUMERIC(5,2),
  tp_4_score NUMERIC(5,2),
  tp_5_score NUMERIC(5,2),
  nilai_akhir NUMERIC(5,2) GENERATED ALWAYS AS ((tp_1_score + tp_2_score + tp_3_score + tp_4_score + tp_5_score) / 5) STORED,
  status_kktp VARCHAR(20),
  deskripsi_rapor TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
